    51→function fmt(t: number): string {
    52→  if (!Number.isFinite(t)) return "0:00";
    53→  const h = Math.floor(t / 3600);
    54→  const m = Math.floor((t % 3600) / 60);
    55→  const s = Math.floor(t % 60);
    56→  return h > 0
    57→    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
    58→    : `${m}:${String(s).padStart(2, "0")}`;
    59→}
